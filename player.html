function buildBestScores(attempts) { bestByRegion = new Map(); const visibleRegionIds = new Set( allRegionsConfig.map(region => region.id) ); const allPlayerAttempts = attempts.filter( row => row.player_id === selectedPlayerId ); const playerAttempts = allPlayerAttempts.filter( row => visibleRegionIds.has(row.region_id) ); const playerName = allPlayerAttempts[0]?.player || selectedPlayerId; document.title = `${playerName} - Regions`; document.getElementById("player-title").textContent = playerName; playerAttempts.forEach(attempt => { const old = bestByRegion.get(attempt.region_id); if (isBetterAttempt(attempt, old)) { bestByRegion.set(attempt.region_id, attempt); } }); }
